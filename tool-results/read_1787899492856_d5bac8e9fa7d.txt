    17→interface ResultPanelProps {
    18→  result: ExtractResult | null;
    19→  loading: boolean;
    20→  /**
    21→   * Called after a snippet is saved as a NEW record (existing behaviour).
    22→   * Receives the freshly-created snippet id so the parent can adopt it as
    23→   * the new `currentSnippetId` (allowing subsequent edits to update the
    24→   * same record in-place).
    25→   */
    26→  onSaved: (snippetId?: string) => void;
    27→  /**
    28→   * ID of the snippet currently loaded into the panel. When set, an extra
    29→   * "Update" button is rendered next to "Simpan" so the user can push
    30→   * their edits back to the same snippet record instead of creating a
    31→   * duplicate.
    32→   */
    33→  currentSnippetId?: string;
    34→  /**
    35→   * Invoked when the user clicks "Update". The parent performs the actual
    36→   * PATCH /api/snippets/[id] call, shows a toast, and refreshes the list.
    37→   * Receives the panel's current (possibly edited) blocks so the PATCH
    38→   * payload reflects in-panel edits (merge, split, delete, duplicate,
    39→   * reorder, text edits) — not the stale `result.blocks` from the parent.
    40→   */
    41→  onUpdateSnippet?: (id: string, blocks: CodeBlock[]) => Promise<void> | void;
    42→}
    43→
    44→export function ResultPanel({ result, loading, onSaved, currentSnippetId, onUpdateSnippet }: ResultPanelProps) {
    45→  const [blocks, setBlocks] = useState<CodeBlock[] | null>(null);
    46→  const [saving, setSaving] = useState(false);
    47→  const [updating, setUpdating] = useState(false);
    48→  const [copiedAll, setCopiedAll] = useState(false);
    49→  const [copiedMd, setCopiedMd] = useState(false);
    50→  const [zipping, setZipping] = useState(false);
    51→  const [exportingHtml, setExportingHtml] = useState(false);
    52→  const [viewMode, setViewMode] = useState<"extracted" | "comparison">("extracted");
    53→  const [reorderMode, setReorderMode] = useState(false);
    54→
    55→  // File extension per language, mirroring the server-side extForLang helper.
    56→  const extForLang = (lang: string): string =>
    57→    ({
    58→      r: "R",
    59→      python: "py",
    60→      sql: "sql",
    61→      java: "java",
    62→      cpp: "cpp",
    63→      javascript: "js",
    64→      typescript: "ts",
    65→      php: "php",
    66→      kotlin: "kt",
    67→      go: "go",
    68→      rust: "rs",
    69→      bash: "sh",
    70→    }[lang] ?? "txt");
    71→
    72→  // BUGFIX: Reset local editable blocks whenever a new result arrives.
    73→  // Previously, editing a block then extracting a new file would show the
    74→  // old edited blocks instead of the fresh extraction result.
    75→  useEffect(() => {
    76→    setBlocks(null);
    77→    setViewMode("extracted");
    78→    setReorderMode(false);
    79→  }, [result]);
    80→
    81→  // Sync local editable blocks whenever a new result arrives.
    82→  const effectiveBlocks = blocks ?? result?.blocks ?? [];
    83→
    84→  const handleSave = async () => {
    85→    if (!result) return;
    86→    setSaving(true);
    87→    try {
    88→      const saved = await saveSnippet(
    89→        result.filename,
    90→        effectiveBlocks,
    91→        effectiveBlocks[0]?.lang ?? "unknown",
    92→        result.size,
    93→      );
    94→      toast.success(`Snippet disimpan · ${saved.id.slice(0, 8)}`);
    95→      // Adopt the freshly-created snippet as the current one so subsequent
    96→      // edits can be pushed back via "Update" rather than re-saving a copy.
    97→      onSaved(saved.id);
    98→    } catch (e: any) {
    99→      toast.error(e?.message ?? "Gagal menyimpan");
   100→    } finally {
   101→      setSaving(false);
   102→    }
   103→  };
   104→
   105→  // Push the current editable blocks back to the *existing* snippet record
   106→  // (identified by `currentSnippetId`). The parent owns the actual API call
   107→  // (so it can also refresh the snippet list + show a toast) — this handler
   108→  // just toggles the loading state on the button and forwards the id.
   109→  const handleUpdate = async () => {
   110→    if (!currentSnippetId || !onUpdateSnippet) return;
   111→    if (effectiveBlocks.length === 0) return;
   112→    setUpdating(true);
   113→    try {
   114→      // Pass the panel's effective (possibly edited) blocks so the parent's
   115→      // PATCH payload reflects in-panel edits — not stale result.blocks.
   116→      await onUpdateSnippet(currentSnippetId, effectiveBlocks);
   117→    } finally {
   118→      setUpdating(false);
   119→    }
   120→  };
   121→
   122→  const handleDownloadAll = () => {
   123→    if (!result) return;
   124→    // Download all blocks as a single text blob (client-side, no snippet needed).
   125→    const dominant = effectiveBlocks[0]?.lang ?? "unknown";
   126→    const ext =
   127→      { r: "R", python: "py", sql: "sql", java: "java", cpp: "cpp", javascript: "js", typescript: "ts" }[dominant] ?? "txt";
   128→    const sep = "#".repeat(60);
   129→    const body = effectiveBlocks
   130→      .map((b, i) => `${sep}\n# Block #${b.index} | ${b.lang} | ${b.lines} lines\n${sep}\n${b.code}`)
   131→      .join("\n\n");
   132→    const blob = new Blob([body], { type: "text/plain;charset=utf-8" });
   133→    const url = URL.createObjectURL(blob);
   134→    const a = document.createElement("a");
   135→    a.href = url;
   136→    a.download = `${result.filename.replace(/\.[^.]+$/, "") || "codelooter"}.${ext}`;
   137→    a.click();
   138→    URL.revokeObjectURL(url);
   139→    toast.success("File kode diunduh");
   140→  };
   141→
   142→  const handleDownloadZip = async () => {
   143→    if (!result || effectiveBlocks.length === 0) return;
   144→    setZipping(true);
   145→    try {
   146→      // Dynamic import keeps jszip out of the initial client bundle.
   147→      const JSZip = (await import("jszip")).default;
   148→      const zip = new JSZip();
   149→      const used = new Map<string, number>();
   150→      for (const b of effectiveBlocks) {
   151→        const ext = extForLang(b.lang);
   152→        let name = `block_${b.index}.${ext}`;
   153→        if (used.has(name)) {
   154→          const n = used.get(name)! + 1;
   155→          used.set(name, n);
   156→          name = `block_${b.index}_${n}.${ext}`;
   157→        } else {
   158→          used.set(name, 0);
   159→        }
   160→        zip.file(name, b.code);
   161→      }
   162→      const blob = await zip.generateAsync({
   163→        type: "blob",
   164→        compression: "DEFLATE",
   165→        compressionOptions: { level: 6 },
   166→      });
   167→      const url = URL.createObjectURL(blob);
   168→      const a = document.createElement("a");
   169→      a.href = url;
   170→      const base = result.filename.replace(/\.[^.]+$/, "") || "codelooter";
   171→      a.download = `${base}_blocks.zip`;
   172→      a.click();
   173→      URL.revokeObjectURL(url);
   174→      toast.success(`ZIP dengan ${effectiveBlocks.length} file dibuat`);
   175→    } catch (e: any) {
   176→      toast.error(e?.message ?? "Gagal membuat ZIP");
   177→    } finally {
   178→      setZipping(false);
   179→    }
   180→  };
   181→
   182→  // Build a self-contained, syntax-highlighted HTML file with all extracted
   183→  // code blocks. Reuses the lightweight `highlight()` tokenizer from
   184→  // code-block-card so the exported colours match the in-app dark-mode view.
   185→  // No external CSS / JS — every style is inlined so the file works offline.
   186→  const handleDownloadHtml = () => {
   187→    if (!result || effectiveBlocks.length === 0) return;
   188→    setExportingHtml(true);
   189→    try {
   190→      // HTML-escape every text node before it lands inside <code>.
   191→      const escapeHtml = (s: string): string =>
   192→        s
   193→          .replace(/&/g, "&amp;")
   194→          .replace(/</g, "&lt;")
   195→          .replace(/>/g, "&gt;")
   196→          .replace(/"/g, "&quot;")
   197→          .replace(/'/g, "&#39;");
   198→
   199→      // Inline-style palette mirroring TOKEN_CLASS dark-mode colours.
   200→      // Typed against `keyof typeof TOKEN_CLASS` so this stays in sync if a
   201→      // new token category is added upstream. Strictly emerald / teal / amber
   202→      // / rose / slate — no blue or indigo.
   203→      const HTML_COLOR: Record<keyof typeof TOKEN_CLASS, { color: string; extra?: string }> = {
   204→        comment: { color: "#94a3b8", extra: "font-style: italic;" }, // slate-400
   205→        string:  { color: "#34d399" },                              // emerald-400
   206→        number:  { color: "#fbbf24" },                              // amber-400
   207→        keyword: { color: "#fb7185", extra: "font-weight: 600;" },  // rose-400
   208→        func:    { color: "#2dd4bf" },                              // teal-400
   209→        ident:   { color: "#e2e8f0" },                              // slate-200
   210→        op:      { color: "#94a3b8" },                              // slate-400
   211→        nl:      { color: "inherit" },
   212→      };
   213→
   214→      // Render a single block's code to highlighted, escaped HTML spans.
   215→      const renderCode = (b: CodeBlock): string => {
   216→        const toks = highlight(b.code, b.lang);
   217→        return toks
   218→          .map((tk) => {
   219→            const esc = escapeHtml(tk.t);
   220→            const cfg = HTML_COLOR[tk.c] ?? HTML_COLOR.ident;
   221→            if (tk.c === "nl") return esc;
   222→            const style = cfg.extra
   223→              ? `color:${cfg.color};${cfg.extra}`
   224→              : `color:${cfg.color}`;
   225→            return `<span style="${style}">${esc}</span>`;
   226→          })
   227→          .join("");
   228→      };
   229→
   230→      const baseName = result.filename.replace(/\.[^.]+$/, "") || "codelooter";
   231→      const today = new Date().toISOString().slice(0, 10);
   232→
   233→      const sections = effectiveBlocks
   234→        .map((b) => {
   235→          const langLabel = (b.lang || "text").toUpperCase();
   236→          return `    <section class="block">
   237→      <header class="block-head">
   238→        <span class="idx">#${b.index}</span>
   239→        <span class="lang">${escapeHtml(langLabel)}</span>
   240→        <span class="meta">${b.lines} lines</span>
   241→      </header>
   242→      <pre><code>${renderCode(b)}</code></pre>
   243→    </section>`;
   244→        })
   245→        .join("\n");
   246→
   247→      const html = `<!DOCTYPE html>
   248→<html lang="en">
   249→<head>
   250→  <meta charset="UTF-8">
   251→  <meta name="viewport" content="width=device-width, initial-scale=1.0">
   252→  <title>${escapeHtml(baseName)} — CodeLooter Export</title>
   253→  <style>
   254→    :root { color-scheme: dark; }
   255→    * { box-sizing: border-box; }
   256→    body {
   257→      margin: 0;
   258→      background: #0f172a;
   259→      color: #e2e8f0;
   260→      font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Helvetica, Arial, sans-serif;
   261→      padding: 1.5rem 1rem 3rem;
   262→      line-height: 1.5;
   263→    }
   264→    .container { max-width: 960px; margin: 0 auto; }
   265→    h1 {
   266→      font-size: 1.5rem;
   267→      font-weight: 700;
   268→      margin: 0 0 .25rem;
   269→      color: #f8fafc;
   270→      word-break: break-all;
   271→    }
   272→    .subtitle {
   273→      color: #94a3b8;
   274→      font-size: .875rem;
   275→      margin: 0 0 1.5rem;
   276→    }
   277→    .block {
   278→      background: #1e293b;
   279→      border: 1px solid #334155;
   280→      border-radius: .6rem;
   281→      margin: 0 0 1rem;
   282→      overflow: hidden;
   283→    }
   284→    .block-head {
   285→      display: flex;
   286→      align-items: center;
   287→      gap: .75rem;
   288→      padding: .6rem .9rem;
   289→      background: #0f172a;
   290→      border-bottom: 1px solid #334155;
   291→      font-size: .8rem;
   292→    }
   293→    .block-head .idx {
   294→      font-family: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;
   295→      color: #94a3b8;
   296→      font-weight: 600;
   297→    }
   298→    .block-head .lang {
   299→      background: rgba(16,185,129,.15);
   300→      color: #6ee7b7;
   301→      padding: .15rem .5rem;
   302→      border-radius: 999px;
   303→      font-size: .7rem;
   304→      font-weight: 600;
   305→      letter-spacing: .03em;
   306→    }
   307→    .block-head .meta {
   308→      color: #64748b;
   309→      font-size: .75rem;
   310→    }
   311→    pre {
   312→      margin: 0;
   313→      padding: 1rem;
   314→      overflow-x: auto;
   315→      background: #0f172a;
   316→    }
   317→    code {
   318→      font-family: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;
   319→      font-size: .82rem;
   320→      line-height: 1.6;
   321→      color: #e2e8f0;
   322→      white-space: pre;
   323→    }
   324→    .footer {
   325→      margin-top: 2rem;
   326→      padding-top: 1rem;
   327→      border-top: 1px solid #334155;
   328→      color: #64748b;
   329→      font-size: .75rem;
   330→      text-align: center;
   331→    }
   332→  </style>
   333→</head>
   334→<body>
   335→  <div class="container">
   336→    <h1>${escapeHtml(result.filename)}</h1>
   337→    <p class="subtitle">${effectiveBlocks.length} blok kode · ${totalLines} baris · dihasilkan oleh CodeLooter</p>
   338→${sections}
   339→    <div class="footer">Diekspor dari CodeLooter — ${today}</div>
   340→  </div>
   341→</body>
   342→</html>`;
   343→
   344→      const blob = new Blob([html], { type: "text/html;charset=utf-8" });
   345→      const url = URL.createObjectURL(blob);
   346→      const a = document.createElement("a");
   347→      a.href = url;
   348→      a.download = `${baseName}_blocks.html`;
   349→      a.click();
   350→      URL.revokeObjectURL(url);
   351→      toast.success(`HTML dengan ${effectiveBlocks.length} blok dibuat`);
   352→    } catch (e: any) {
   353→      toast.error(e?.message ?? "Gagal membuat HTML");
   354→    } finally {
   355→      setExportingHtml(false);
   356→    }
   357→  };
   358→
   359→  const handleBlockDownload = (index: number) => {
   360→    const b = effectiveBlocks.find((x) => x.index === index);
   361→    if (!b) return;
   362→    const ext =
   363→      { r: "R", python: "py", sql: "sql", java: "java", cpp: "cpp", javascript: "js", typescript: "ts" }[b.lang] ?? "txt";
   364→    const blob = new Blob([b.code], { type: "text/plain;charset=utf-8" });
   365→    const url = URL.createObjectURL(blob);
   366→    const a = document.createElement("a");
   367→    a.href = url;
   368→    a.download = `block_${index}.${ext}`;
   369→    a.click();
   370→    URL.revokeObjectURL(url);
   371→  };
   372→
   373→  const handleBlockChange = (index: number, code: string) => {
   374→    setBlocks((prev) => {
   375→      const base = prev ?? result?.blocks ?? [];
   376→      return base.map((b) => (b.index === index ? { ...b, code, lines: code.split("\n").length } : b));
   377→    });
   378→  };
   379→
   380→  // Reorder blocks via drag-and-drop. The new order is committed to local
   381→  // state and block indices are renumbered to match the new positions.
   382→  const handleReorder = (reordered: CodeBlock[]) => {
   383→    setBlocks(reordered.map((b, i) => ({ ...b, index: i })));
   384→    toast.info("Urutan blok diperbarui");
   385→  };
   386→
   387→  // Merge block at `index` with the next block. The two blocks' code is
   388→  // concatenated with a newline, and indices are renumbered.
   389→  const handleMergeWithNext = (index: number) => {
   390→    setBlocks((prev) => {
   391→      const base = prev ?? result?.blocks ?? [];
   392→      const idx = base.findIndex((b) => b.index === index);
   393→      if (idx < 0 || idx >= base.length - 1) return base;
   394→      const merged = {
   395→        ...base[idx],
   396→        code: base[idx].code + "\n" + base[idx + 1].code,
   397→        lines: base[idx].lines + base[idx + 1].lines,
   398→        source: base[idx].source === base[idx + 1].source
   399→          ? base[idx].source
   400→          : `${base[idx].source}+merge`,
   401→      };
   402→      const next = [...base.slice(0, idx), merged, ...base.slice(idx + 2)];
   403→      return next.map((b, i) => ({ ...b, index: i }));
   404→    });
   405→  };
   406→
   407→  // Split block at `index` into two blocks at line `atLine`. Lines 1..atLine-1
   408→  // stay in the first block, lines atLine..end go to a new second block.
   409→  const handleSplit = (index: number, atLine: number) => {
   410→    setBlocks((prev) => {
   411→      const base = prev ?? result?.blocks ?? [];
   412→      const idx = base.findIndex((b) => b.index === index);
   413→      if (idx < 0) return base;
   414→      const lines = base[idx].code.split("\n");
   415→      if (atLine < 2 || atLine >= lines.length) return base;
   416→      const firstCode = lines.slice(0, atLine).join("\n");
   417→      const secondCode = lines.slice(atLine).join("\n");
   418→      const first = { ...base[idx], code: firstCode, lines: firstCode.split("\n").length };
   419→      const second = { ...base[idx], code: secondCode, lines: secondCode.split("\n").length, source: "split" };
   420→      const next = [...base.slice(0, idx), first, second, ...base.slice(idx + 1)];
   421→      return next.map((b, i) => ({ ...b, index: i }));
   422→    });
   423→  };
   424→
   425→  // Delete block at `index`. Indices are renumbered.
   426→  const handleDelete = (index: number) => {
   427→    setBlocks((prev) => {
   428→      const base = prev ?? result?.blocks ?? [];
   429→      const next = base.filter((b) => b.index !== index);
   430→      return next.map((b, i) => ({ ...b, index: i }));
   431→    });
   432→  };
   433→
   434→  // Duplicate block at `index`. The copy is inserted right after the original.
   435→  const handleDuplicate = (index: number) => {
   436→    setBlocks((prev) => {
   437→      const base = prev ?? result?.blocks ?? [];
   438→      const idx = base.findIndex((b) => b.index === index);
   439→      if (idx < 0) return base;
   440→      const copy = { ...base[idx], source: "duplicate" };
   441→      const next = [...base.slice(0, idx + 1), copy, ...base.slice(idx + 1)];
   442→      return next.map((b, i) => ({ ...b, index: i }));
   443→    });
   444→  };
   445→
   446→  const handleCopyAll = async () => {
   447→    const body = effectiveBlocks.map((b) => b.code).join("\n\n");
   448→    try {
   449→      await navigator.clipboard.writeText(body);
   450→      setCopiedAll(true);
   451→      setTimeout(() => setCopiedAll(false), 1500);
   452→      toast.success(`${effectiveBlocks.length} blok disalin ke clipboard`);
   453→    } catch {
   454→      toast.error("Gagal menyalin");
   455→    }
   456→  };
   457→
   458→  // Copy all blocks to clipboard as Markdown fenced code blocks,
   459→  // each preceded by an HTML comment header with index / language / line count.
   460→  // Example output:
   461→  //   <!-- Block #0 | r | 5 lines -->
   462→  //   ```r
   463→  //   library(ggplot2)
   464→  //   ```
   465→  const handleCopyMarkdown = async () => {
   466→    const body = effectiveBlocks
   467→      .map(
   468→        (b) =>
   469→          `<!-- Block #${b.index} | ${b.lang} | ${b.lines} lines -->\n\`\`\`${b.lang}\n${b.code}\n\`\`\``,
   470→      )
   471→      .join("\n\n");
   472→    try {
   473→      await navigator.clipboard.writeText(body);
   474→      setCopiedMd(true);
   475→      setTimeout(() => setCopiedMd(false), 1500);
   476→      toast.success(`${effectiveBlocks.length} blok disalin sebagai Markdown`);
   477→    } catch {
   478→      toast.error("Gagal menyalin");
   479→    }
   480→  };
   481→
   482→  // Language distribution for the summary bar.
   483→  const langDist = effectiveBlocks.reduce<Record<string, number>>((acc, b) => {
   484→    acc[b.lang] = (acc[b.lang] || 0) + 1;
   485→    return acc;
   486→  }, {});
   487→  const totalLines = effectiveBlocks.reduce((sum, b) => sum + b.lines, 0);
   488→  const totalChars = effectiveBlocks.reduce((sum, b) => sum + b.code.length, 0);
   489→
   490→  if (loading) {
   491→    return <LoadingSkeleton />;
   492→  }
   493→
   494→  if (!result) {
   495→    return (
   496→      <div className="flex h-full min-h-[400px] flex-col items-center justify-center gap-3 rounded-xl border border-dashed border-border bg-muted/20 p-8 text-center">
   497→        <div className="flex h-14 w-14 items-center justify-center rounded-full bg-muted text-muted-foreground">
   498→          <Code2 className="h-7 w-7" />
   499→        </div>
   500→        <p className="text-sm font-medium">Hasil ekstraksi akan muncul di sini</p>
   501→        <p className="max-w-xs text-xs text-muted-foreground">
   502→          Upload file praktikum / paper, pilih bahasa, lalu klik{" "}
   503→          <span className="font-mono text-emerald-600 dark:text-emerald-400">Ekstrak kode</span>.
   504→        </p>
   505→      </div>
   506→    );
   507→  }
   508→
   509→  return (
   510→    <div className="flex flex-col gap-4">
   511→      {/* File header */}
   512→      <motion.div
   513→        initial={{ opacity: 0, y: -8 }}
   514→        animate={{ opacity: 1, y: 0 }}
   515→        transition={{ duration: 0.25 }}

   525→              {(result.size / 1024).toFixed(1)} KB · {result.total} blok kode · {totalLines} baris · {totalChars.toLocaleString()} karakter

   540→              size="sm"
   541→              variant="ghost"
   542→              onClick={handleCopyMarkdown}
   543→              disabled={effectiveBlocks.length === 0}
   544→              title="Salin semua blok sebagai Markdown"

   550→              size="sm"

   569→              size="sm"

   571→              onClick={handleDownloadHtml}

   587→                <GitCompare className="h-4 w-4" />

   605→              size="sm"

   615→                size="sm"
   616→                onClick={handleUpdate}

   626→        </div>

   633→                key={lang}
   634→                className="inline-flex items-center gap-1 rounded-full bg-emerald-500/10 px-2 py-0.5 text-[11px] font-medium text-emerald-700 dark:text-emerald-300"

   641→        {result.stats && <StatsBar stats={result.stats} />}
   642→        {result.stats && <StatsChart stats={result.stats} />}
   643→      </motion.div>
   644→
   645→      {/* Blocks or Comparison view */}
   646→      {viewMode === "comparison" && result.stats?.removedLines ? (

   648→      ) : effectiveBlocks.length === 0 ? (
   649→        <div className="rounded-xl border border-amber-500/30 bg-amber-500/5 p-6 text-center">
   650→          <Boxes className="mx-auto mb-2 h-8 w-8 text-amber-600" />
   651→          <p className="text-sm font-medium">Tidak ada blok kode terdeteksi</p>
   652→          <p className="mt-1 text-xs text-muted-foreground">
   653→            File mungkin berbasis gambar (perlu OCR) atau tidak memiliki marker kode.
   654→          </p>
   655→        </div>
   656→      ) : reorderMode ? (
   657→        <SortableBlockList
   658→          blocks={effectiveBlocks}
   659→          onReorder={handleReorder}
   660→          onDownload={handleBlockDownload}
   661→          onChange={handleBlockChange}
   662→          onMergeWithNext={handleMergeWithNext}
   663→          onSplit={handleSplit}
   664→          onDelete={handleDelete}
   665→          onDuplicate={handleDuplicate}
   666→        />
   667→      ) : (
   668→        <AnimatePresence mode="popLayout">
   669→          {effectiveBlocks.map((b, i) => (
   670→            <motion.div
   671→              key={`${result.filename}-${b.index}`}
   672→              layout
   673→              initial={{ opacity: 0, y: 12 }}
   674→              animate={{ opacity: 1, y: 0 }}
   675→              exit={{ opacity: 0, y: -8 }}
   676→              transition={{ duration: 0.2, delay: b.index * 0.04 }}
   677→            >
   678→              <CodeBlockCard
   679→                block={b}
   680→                onDownload={handleBlockDownload}
   681→                onChange={handleBlockChange}
   682→                onMergeWithNext={handleMergeWithNext}
   683→                onSplit={handleSplit}
   684→                onDelete={handleDelete}
   685→                onDuplicate={handleDuplicate}
   686→                isLast={i === effectiveBlocks.length - 1}
   687→              />
   688→            </motion.div>
   689→          ))}
   690→        </AnimatePresence>
   691→      )}
   692→    </div>
