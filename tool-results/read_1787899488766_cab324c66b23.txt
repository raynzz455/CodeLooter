    23→    23→   * the new `currentSnippetId` (allowing subsequent edits to update the
    24→    24→   * same record in-place).
    25→    25→   */
    26→    26→  onSaved: (snippetId?: string) => void;
    27→    27→  /**
    28→    28→   * ID of the snippet currently loaded into the panel. When set, an extra

    40→    40→   */
    41→    41→  onUpdateSnippet?: (id: string, blocks: CodeBlock[]) => Promise<void> | void;

   389→   389→  const handleMergeWithNext = (index: number) => {

   391→   391→      const base = prev ?? result?.blocks ?? [];
   392→   392→      const idx = base.findIndex((b) => b.index === index);
   393→   393→      if (idx < 0 || idx >= base.length - 1) return base;
   394→   394→      const merged = {
   395→   395→        ...base[idx],
   396→   396→        code: base[idx].code + "\n" + base[idx + 1].code,
   397→   397→        lines: base[idx].lines + base[idx + 1].lines,
   398→   398→        source: base[idx].source === base[idx + 1].source
   399→   399→          ? base[idx].source
   400→   400→          : `${base[idx].source}+merge`,
   401→   401→      };
   402→   402→      const next = [...base.slice(0, idx), merged, ...base.slice(idx + 2)];
   403→   403→      return next.map((b, i) => ({ ...b, index: i }));
   404→   404→    });

   411→   411→      const base = prev ?? result?.blocks ?? [];
   412→   412→      const idx = base.findIndex((b) => b.index === index);

   415→   415→      if (atLine < 2 || atLine >= lines.length) return base;
   416→   416→      const firstCode = lines.slice(0, atLine).join("\n");
   417→   417→      const secondCode = lines.slice(atLine).join("\n");
   418→   418→      const first = { ...base[idx], code: firstCode, lines: firstCode.split("\n").length };
   419→   419→      const second = { ...base[idx], code: secondCode, lines: secondCode.split("\n").length, source: "split" };
   420→   420→      const next = [...base.slice(0, idx), first, second, ...base.slice(idx + 1)];
   421→   421→      return next.map((b, i) => ({ ...b, index: i }));
   422→   422→    });
   423→   423→  };

   431→   431→    });
   432→   432→  };
   433→   433→
   434→   434→  // Duplicate block at `index`. The copy is inserted right after the original.
   435→   435→  const handleDuplicate = (index: number) => {
   436→   436→    setBlocks((prev) => {
   437→   437→      const base = prev ?? result?.blocks ?? [];
   438→   438→      const idx = base.findIndex((b) => b.index === index);
   439→   439→      if (idx < 0) return base;
   440→   440→      const copy = { ...base[idx], source: "duplicate" };
   441→   441→      const next = [...base.slice(0, idx + 1), copy, ...base.slice(idx + 1)];
   442→   442→      return next.map((b, i) => ({ ...b, index: i }));
   443→   443→    });
   444→   444→  };
   445→   445→
   446→   446→  const handleCopyAll = async () => {
   447→   447→    const body = effectiveBlocks.map((b) => b.code).join("\n\n");
   448→   448→    try {
   449→   449→      await navigator.clipboard.writeText(body);
   450→   450→      setCopiedAll(true);
   451→   451→      setTimeout(() => setCopiedAll(false), 1500);
   452→   452→      toast.success(`${effectiveBlocks.length} blok disalin ke clipboard`);
   453→   453→    } catch {
   454→   454→      toast.error("Gagal menyalin");
   455→   455→    }
   456→   456→  };
   457→   457→
   458→   458→  // Copy all blocks to clipboard as Markdown fenced code blocks,
   459→   459→  // each preceded by an HTML comment header with index / language / line count.
   460→   460→  // Example output:
   461→   461→  //   <!-- Block #0 | r | 5 lines -->
   462→   462→  //   ```r
   463→   463→  //   library(ggplot2)
   464→   464→  //   ```
   465→   465→  const handleCopyMarkdown = async () => {
   466→   466→    const body = effectiveBlocks
   467→   467→      .map(
   468→   468→        (b) =>
   469→   469→          `<!-- Block #${b.index} | ${b.lang} | ${b.lines} lines -->\n\`\`\`${b.lang}\n${b.code}\n\`\`\``,
   470→   470→      )
   471→   471→      .join("\n\n");
   472→   472→    try {
   473→   473→      await navigator.clipboard.writeText(body);
   474→   474→      setCopiedMd(true);
   475→   475→      setTimeout(() => setCopiedMd(false), 1500);
   476→   476→      toast.success(`${effectiveBlocks.length} blok disalin sebagai Markdown`);
   477→   477→    } catch {
   478→   478→      toast.error("Gagal menyalin");
   479→   479→    }
   480→   480→  };
   481→   481→
   482→   482→  // Language distribution for the summary bar.
   483→   483→  const langDist = effectiveBlocks.reduce<Record<string, number>>((acc, b) => {
   484→   484→    acc[b.lang] = (acc[b.lang] || 0) + 1;
   485→   485→    return acc;
   486→   486→  }, {});
   487→   487→  const totalLines = effectiveBlocks.reduce((sum, b) => sum + b.lines, 0);
   488→   488→  const totalChars = effectiveBlocks.reduce((sum, b) => sum + b.code.length, 0);
   489→   489→
   490→   490→  if (loading) {
   491→   491→    return <LoadingSkeleton />;
   492→   492→  }
   493→   493→
   494→   494→  if (!result) {
   495→   495→    return (
   496→   496→      <div className="flex h-full min-h-[400px] flex-col items-center justify-center gap-3 rounded-xl border border-dashed border-border bg-muted/20 p-8 text-center">
   497→   497→        <div className="flex h-14 w-14 items-center justify-center rounded-full bg-muted text-muted-foreground">
   498→   498→          <Code2 className="h-7 w-7" />
   499→   499→        </div>
   500→   500→        <p className="text-sm font-medium">Hasil ekstraksi akan muncul di sini</p>
   501→   501→        <p className="max-w-xs text-xs text-muted-foreground">
   502→   502→          Upload file praktikum / paper, pilih bahasa, lalu klik{" "}
   503→   503→          <span className="font-mono text-emerald-600 dark:text-emerald-400">Ekstrak kode</span>.
   504→   504→        </p>
   505→   505→      </div>
   506→   506→    );
   507→   507→  }
   508→   508→
   509→   509→  return (
   510→   510→    <div className="flex flex-col gap-4">
   511→   511→      {/* File header */}
   512→   512→      <motion.div
   513→   513→        initial={{ opacity: 0, y: -8 }}
   514→   514→        animate={{ opacity: 1, y: 0 }}
   515→   515→        transition={{ duration: 0.25 }}
   516→   516→        className="flex flex-col gap-3 rounded-xl border border-border bg-card p-4 shadow-sm"
   517→   517→      >
   518→   518→        <div className="flex items-start gap-3">
   519→   519→          <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-emerald-500/10 text-emerald-600 dark:text-emerald-400">
   520→   520→            <FileText className="h-5 w-5" />
   521→   521→          </div>
   522→   522→          <div className="min-w-0 flex-1">
   523→   523→            <p className="truncate font-mono text-sm font-semibold">{result.filename}</p>
   524→   524→            <p className="text-xs text-muted-foreground">
   525→   525→              {(result.size / 1024).toFixed(1)} KB · {result.total} blok kode · {totalLines} baris · {totalChars.toLocaleString()} karakter
   526→   526→            </p>
   527→   527→          </div>
   528→   528→          <div className="flex shrink-0 gap-2">
   529→   529→            <Button
   530→   530→              size="sm"
   531→   531→              variant="ghost"
   532→   532→              onClick={handleCopyAll}
   533→   533→              disabled={effectiveBlocks.length === 0}
   534→   534→              title="Salin semua blok"
   535→   535→            >
   536→   536→              {copiedAll ? <Check className="h-4 w-4 text-emerald-600" /> : <Copy className="h-4 w-4" />}
   537→   537→              <span className="hidden sm:inline">Salin semua</span>
   538→   538→            </Button>
   539→   539→            <Button
   540→   540→              size="sm"
   541→   541→              variant="ghost"
   542→   542→              onClick={handleCopyMarkdown}
   543→   543→              disabled={effectiveBlocks.length === 0}
   544→   544→              title="Salin semua blok sebagai Markdown"
   545→   545→            >
   546→   546→              {copiedMd ? <Check className="h-4 w-4 text-emerald-600" /> : <ClipboardCopy className="h-4 w-4" />}
   547→   547→              <span className="hidden sm:inline">Markdown</span>
   548→   548→            </Button>
   549→   549→            <Button
   550→   550→              size="sm"
   551→   551→              variant="outline"
   552→   552→              onClick={handleDownloadAll}
   553→   553→              disabled={effectiveBlocks.length === 0}
   554→   554→            >
   555→   555→              <Download className="h-4 w-4" />
   556→   556→              <span className="hidden sm:inline">Download</span>
   557→   557→            </Button>
   558→   558→            <Button
   559→   559→              size="sm"
   560→   560→              variant="outline"
   561→   561→              onClick={handleDownloadZip}
   562→   562→              disabled={effectiveBlocks.length === 0 || zipping}
   563→   563→              title="Download semua blok sebagai file ZIP terpisah"
   564→   564→            >
   565→   565→              {zipping ? <Loader2 className="h-4 w-4 animate-spin" /> : <FileArchive className="h-4 w-4" />}
   566→   566→              <span className="hidden sm:inline">{zipping ? "Zipping…" : "ZIP"}</span>
   567→   567→            </Button>
   568→   568→            <Button
   569→   569→              size="sm"
   570→   570→              variant="outline"
   571→   571→              onClick={handleDownloadHtml}
   572→   572→              disabled={effectiveBlocks.length === 0 || exportingHtml}
   573→   573→              title="Download semua blok sebagai file HTML syntax-highlighted"
   574→   574→            >
   575→   575→              {exportingHtml ? <Loader2 className="h-4 w-4 animate-spin" /> : <FileCode2 className="h-4 w-4" />}
   576→   576→              <span className="hidden sm:inline">{exportingHtml ? "Membuat…" : "HTML"}</span>
   577→   577→            </Button>
   578→   578→            {/* Comparison view toggle — only show if removedLines exist */}
   579→   579→            {result.stats?.removedLines && result.stats.removedLines.length > 0 && (
   580→   580→              <Button
   581→   581→                size="sm"
   582→   582→                variant={viewMode === "comparison" ? "default" : "outline"}
   583→   583→                onClick={() => setViewMode((v) => v === "comparison" ? "extracted" : "comparison")}
   584→   584→                title="Bandingkan kode yang diekstrak vs baris yang dihapus"
   585→   585→                className={viewMode === "comparison" ? "bg-rose-600 hover:bg-rose-700" : ""}
   586→   586→              >
   587→   587→                <GitCompare className="h-4 w-4" />
   588→   588→                <span className="hidden sm:inline">Bandingkan</span>
   589→   589→              </Button>
   590→   590→            )}
   591→   591→            {/* Reorder toggle — only in extracted view */}
   592→   592→            {viewMode === "extracted" && effectiveBlocks.length > 1 && (
   593→   593→              <Button
   594→   594→                size="sm"
   595→   595→                variant={reorderMode ? "default" : "outline"}
   596→   596→                onClick={() => setReorderMode((r) => !r)}
   597→   597→                title={reorderMode ? "Selesai mengurutkan" : "Urutkan ulang blok (drag & drop)"}
   598→   598→                className={reorderMode ? "bg-teal-600 hover:bg-teal-700" : ""}
   599→   599→              >
   600→   600→                <ArrowUpDown className="h-4 w-4" />
   601→   601→                <span className="hidden sm:inline">{reorderMode ? "Selesai" : "Urutkan"}</span>
   602→   602→              </Button>
   603→   603→            )}
   604→   604→            <Button
   605→   605→              size="sm"
   606→   606→              onClick={handleSave}
   607→   607→              disabled={saving || effectiveBlocks.length === 0}
   608→   608→              className="bg-emerald-600 hover:bg-emerald-700"
   609→   609→            >
   610→   610→              {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />}
   611→   611→              <span className="hidden sm:inline">Simpan</span>
   612→   612→            </Button>
   613→   613→            {currentSnippetId && onUpdateSnippet && (
   614→   614→              <Button
   615→   615→                size="sm"
   616→   616→                onClick={handleUpdate}
   617→   617→                disabled={updating || effectiveBlocks.length === 0}
   618→   618→                className="bg-amber-600 hover:bg-amber-700"
   619→   619→                title={`Perbarui snippet ini (${currentSnippetId.slice(0, 8)}) di tempat`}
   620→   620→              >
   621→   621→                {updating ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />}
   622→   622→                <span className="hidden sm:inline">{updating ? "Memperbarui…" : "Update"}</span>
   623→   623→              </Button>
   624→   624→            )}
   625→   625→          </div>
   626→   626→        </div>
   627→   627→        {/* Language distribution badges */}
   628→   628→        {effectiveBlocks.length > 0 && (
   629→   629→          <div className="flex flex-wrap items-center gap-1.5 border-t border-border/40 pt-2">
   630→   630→            <span className="text-[11px] text-muted-foreground">Bahasa:</span>
   631→   631→            {Object.entries(langDist).map(([lang, count]) => (
   632→   632→              <span
   633→   633→                key={lang}
   634→   634→                className="inline-flex items-center gap-1 rounded-full bg-emerald-500/10 px-2 py-0.5 text-[11px] font-medium text-emerald-700 dark:text-emerald-300"
   635→   635→              >
   636→   636→                {lang} ×{count}
   637→   637→              </span>
   638→   638→            ))}
   639→   639→          </div>
   640→   640→        )}
   641→   641→        {result.stats && <StatsBar stats={result.stats} />}
   642→   642→        {result.stats && <StatsChart stats={result.stats} />}
   643→   643→      </motion.div>
   644→   644→
   645→   645→      {/* Blocks or Comparison view */}
   646→   646→      {viewMode === "comparison" && result.stats?.removedLines ? (
   647→   647→        <ComparisonView blocks={effectiveBlocks} removedLines={result.stats.removedLines} />
   648→   648→      ) : effectiveBlocks.length === 0 ? (
   649→   649→        <div className="rounded-xl border border-amber-500/30 bg-amber-500/5 p-6 text-center">
   650→   650→          <Boxes className="mx-auto mb-2 h-8 w-8 text-amber-600" />
   651→   651→          <p className="text-sm font-medium">Tidak ada blok kode terdeteksi</p>
   652→   652→          <p className="mt-1 text-xs text-muted-foreground">
   653→   653→            File mungkin berbasis gambar (perlu OCR) atau tidak memiliki marker kode.
   654→   654→          </p>
   655→   655→        </div>
   656→   656→      ) : reorderMode ? (
   657→   657→        <SortableBlockList
   658→   658→          blocks={effectiveBlocks}
   659→   659→          onReorder={handleReorder}
   660→   660→          onDownload={handleBlockDownload}
   661→   661→          onChange={handleBlockChange}
   662→   662→          onMergeWithNext={handleMergeWithNext}
   663→   663→          onSplit={handleSplit}
   664→   664→          onDelete={handleDelete}
   665→   665→          onDuplicate={handleDuplicate}
   666→   666→        />
   667→   667→      ) : (
   668→   668→        <AnimatePresence mode="popLayout">
   669→   669→          {effectiveBlocks.map((b, i) => (
   670→   670→            <motion.div
   671→   671→              key={`${result.filename}-${b.index}`}
   672→   672→              layout
   673→   673→              initial={{ opacity: 0, y: 12 }}
   674→   674→              animate={{ opacity: 1, y: 0 }}
   675→   675→              exit={{ opacity: 0, y: -8 }}
   676→   676→              transition={{ duration: 0.2, delay: b.index * 0.04 }}
   677→   677→            >
   678→   678→              <CodeBlockCard
   679→   679→                block={b}
   680→   680→                onDownload={handleBlockDownload}
   681→   681→                onChange={handleBlockChange}
   682→   682→                onMergeWithNext={handleMergeWithNext}
   683→   683→                onSplit={handleSplit}
   684→   684→                onDelete={handleDelete}
   685→   685→                onDuplicate={handleDuplicate}
   686→   686→                isLast={i === effectiveBlocks.length - 1}
   687→   687→              />
   688→   688→            </motion.div>
   689→   689→          ))}
   690→   690→        </AnimatePresence>
   691→   691→      )}
   692→   692→    </div>
   693→   693→  );
   694→   694→}
