   438→   438→    setBlocks((prev) => {
   439→   439→      const base = prev ?? result?.blocks ?? [];

   441→   441→      if (idx < 0) return base;

   443→   443→      if (atLine < 2 || atLine >= lines.length) return base;
   444→   444→      const firstCode = lines.slice(0, atLine).join("\n");
   445→   445→      const secondCode = lines.slice(atLine).join("\n");
   446→   446→      const first = { ...base[idx], code: firstCode, lines: firstCode.split("\n").length };
   447→   447→      const second = { ...base[idx], code: secondCode, lines: secondCode.split("\n").length, source: "split" };
   448→   448→      const next = [...base.slice(0, idx), first, second, ...base.slice(idx + 1)];
   449→   449→      return next.map((b, i) => ({ ...b, index: i }));
   450→   450→    });
   451→   451→  };

   453→   453→  // Delete block at `index`. Indices are renumbered.

   458→   458→      return next.map((b, i) => ({ ...b, index: i }));
   459→   459→    });
   460→   460→  };

   462→   462→  // Duplicate block at `index`. The copy is inserted right after the original.
   463→   463→  const handleDuplicate = (index: number) => {

   465→   465→      const base = prev ?? result?.blocks ?? [];
   466→   466→      const idx = base.findIndex((b) => b.index === index);
   467→   467→      if (idx < 0) return base;
   468→   468→      const copy = { ...base[idx], source: "duplicate" };
   469→   469→      const next = [...base.slice(0, idx + 1), copy, ...base.slice(idx + 1)];
   470→   470→      return next.map((b, i) => ({ ...b, index: i }));
   471→   471→    });
   472→   472→  };
   473→   473→
