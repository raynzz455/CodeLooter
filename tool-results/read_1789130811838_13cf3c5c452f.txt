     4→//
     5→// Adapted from the original CodeLooter repo's app/page.tsx:
     6→//   - Removed auth/user profile (no login in our single-user sandbox).
     7→//   - Removed useRouter navigation — this is a single-page app.

     9→//   - extractCode → extractFile (pattern-based, 100% free, no LLM).
    10→//   - saveSnippet signature now takes (filename, blocks, lang, size, tags).
    11→//   - Added a "SNIPPET TERSIMPAN" section below the main grid that lists
    12→//     saved snippets and lets the user load one back into the result panel
    13→//     or delete it.

    50→function Tag({
    51→  bg = "#ffe8a3",
    52→  color = "#000",
    53→  children,
    54→}: {
    55→  bg?: string;
    56→  color?: string;
    57→  children: React.ReactNode;
    58→}) {
    59→  return (
    60→    <span
    61→      style={{
    62→        backgroundColor: bg,
    63→        color,
    64→        border: "2px solid #000",
    65→        borderRadius: "6px",
    66→        padding: "2px 9px",
    67→        fontSize: "0.68rem",
    68→        fontWeight: 900,
    69→        whiteSpace: "nowrap",
    70→        fontFamily: "var(--font-body)",
    71→      }}
    72→    >
    73→      {children}
    74→    </span>
    75→  );
    76→}

   131→function LangChip({

   137→  active: boolean;

   148→        fontSize: "0.75rem",

   166→// ─── Main page ───
   167→export default function Home() {
   168→  // Language picker state.
   169→  const [selectedLang, setSelectedLang] = useState<string>("r");
   170→  const [dropdownOpen, setDropdownOpen] = useState(false);
   171→
   172→  // Upload state.
   173→  const [uploadedFile, setUploadedFile] = useState<File | null>(null);
   174→  const [isDragging, setIsDragging] = useState(false);
   175→  const fileInputRef = useRef<HTMLInputElement>(null);
   176→
   177→  // Extraction state.
   178→  const [isExtracting, setIsExtracting] = useState(false);
   179→  const [extractedBlocks, setExtractedBlocks] = useState<CodeBlock[]>([]);
   180→  const [extractError, setExtractError] = useState<string | null>(null);
   181→  const [extractMeta, setExtractMeta] = useState<{
   182→    method?: string;
   183→    durationMs?: number;
   184→    cached?: boolean;
   185→  }>({});
   186→
   187→  // Per-result actions.
   188→  const [copied, setCopied] = useState(false);
   189→  const [saving, setSaving] = useState(false);
   190→  const [savedSnippetId, setSavedSnippetId] = useState<string | null>(null);
   191→
   192→  // Snippet list (sidebar / footer section).
   193→  const [snippets, setSnippets] = useState<SnippetMeta[]>([]);
   194→  const [loadingSnippets, setLoadingSnippets] = useState(true);
   195→
   196→  const langDropdownRef = useRef<HTMLDivElement>(null);
   197→
   198→  const extracted = extractedBlocks.length > 0;
   199→  const currentLang =
   200→    LANGUAGES.find((l) => l.id === selectedLang) ?? LANGUAGES[0];
   201→
   202→  // active block for preview — prefers the user-selected language, else the
   203→  // first non-"unknown" block, else the first block, else falls back to the
   204→  // sample code so the panel is never empty.
   205→  const activeBlock =
   206→    extractedBlocks.find((b) => b.lang === selectedLang) ??
   207→    extractedBlocks.find((b) => b.lang !== "unknown") ??
   208→    extractedBlocks[0] ??
   209→    null;
   210→  const displayCode = activeBlock?.code ?? SAMPLE_CODES[selectedLang] ?? "";
   211→
   212→  // Detected languages across all extracted blocks.
   213→  const detectedLangs = Array.from(
   214→    new Set(extractedBlocks.map((b) => b.lang)),
   215→  ).filter((l) => LANGUAGES.some((lang) => lang.id === l));
   216→
   217→  // ─── Load saved snippets from /api/snippets on mount ───
   218→  const refreshSnippets = useCallback(async () => {
   219→    setLoadingSnippets(true);
   220→    try {
   221→      const list = await listSnippets();
   222→      setSnippets(list);
   223→    } catch (err) {
   224→      // Don't toast on initial load — just log to console.
   225→      console.error("Failed to load snippets:", err);
   226→    } finally {
   227→      setLoadingSnippets(false);
   228→    }
   229→  }, []);
   230→
   231→  useEffect(() => {
   232→    void refreshSnippets();
   233→  }, [refreshSnippets]);
   234→
   235→  // Close language dropdown on outside click.
   236→  useEffect(() => {
   237→    const handler = (e: MouseEvent) => {
   238→      if (
   239→        langDropdownRef.current &&
   240→        !langDropdownRef.current.contains(e.target as Node)
   241→      ) {
   242→        setDropdownOpen(false);
   243→      }
   244→    };
   245→    document.addEventListener("mousedown", handler);
   246→    return () => document.removeEventListener("mousedown", handler);
   247→  }, []);
   248→
   249→  // ─── File handling ───
   250→  const setFile = (file: File | null) => {
   251→    setUploadedFile(file);
   252→    setExtractedBlocks([]);
   253→    setExtractError(null);
   254→    setSavedSnippetId(null);
   255→    setExtractMeta({});
   256→  };
   257→
   258→  const handleDrop = useCallback((e: React.DragEvent) => {
   259→    e.preventDefault();
   260→    setIsDragging(false);
   261→    const file = e.dataTransfer.files[0];
   262→    if (file && ACCEPTED_EXT.test(file.name)) setFile(file);
   263→    else if (file)
   264→      toast.error("Format file tidak didukung", {
   265→        description: file.name,
   266→      });
   267→  }, []);
   268→
   269→  const handleFileInput = (e: React.ChangeEvent<HTMLInputElement>) => {
   270→    const file = e.target.files?.[0];
   271→    if (file) setFile(file);
   272→  };
   273→
   274→  // ─── Extract ───
   275→  // 100% pattern-based extraction. No LLM, no external API.
   276→  const handleExtract = async () => {
   277→    if (!uploadedFile) return;
   278→    setIsExtracting(true);
   279→    setExtractError(null);
   280→    setExtractedBlocks([]);
   281→    setSavedSnippetId(null);
   282→    setExtractMeta({});
   283→
   284→    try {
   285→      const data: ExtractResult = await extractFile(uploadedFile, selectedLang);
   286→
   287→      const blocks = data.blocks ?? [];
   288→      setExtractedBlocks(blocks);
   289→      setExtractMeta({
   290→        method: data.stats?.method,
   291→        durationMs: data.stats?.durationMs,
   292→        cached: data.cached,
   293→      });
   294→
   295→      if (blocks.length > 0) {
   296→        const firstKnown =
   297→          blocks.find((b) => b.lang !== "unknown") ?? blocks[0];
   298→        setSelectedLang(firstKnown.lang);
   299→        toast.success(
   300→          `${blocks.length} blok kode berhasil diekstrak!${data.cached ? " (cache)" : ""}`,
   301→          {
   302→            description: `⚡ ${data.stats?.method ?? "pattern"}`,
   303→          },
   304→        );
   305→      } else {
   306→        toast.warning("Tidak ada blok kode terdeteksi", {
   307→          description: "Coba bahasa lain atau periksa format file.",
   308→        });
   309→      }
   310→    } catch (err) {
   311→      const msg =
   312→        err instanceof Error ? err.message : "Koneksi gagal — coba lagi";
   313→      setExtractError(msg);
   314→      toast.error("Ekstraksi gagal", { description: msg });
   315→    } finally {
   316→      setIsExtracting(false);
   317→    }
   318→  };
   319→
   320→  // ─── Save snippet ───
   321→  // Persists the current extracted blocks to the local SQLite store via
   322→  // /api/snippets. On success, refreshes the snippet list and shows a "saved"
   323→  // state on the result panel's save button.
   324→  const handleSave = async () => {
   325→    if (!uploadedFile || extractedBlocks.length === 0) return;
   326→    setSaving(true);
   327→    try {
   328→      const { id } = await saveSnippet(
   329→        uploadedFile.name,
   330→        extractedBlocks,
   331→        selectedLang,
   332→        uploadedFile.size,
   333→        "",
   334→      );
   335→      setSavedSnippetId(id);
   336→      toast.success("Snippet tersimpan!", {
   337→        description: `${extractedBlocks.length} blok · ${uploadedFile.name}`,
   338→      });
   339→      void refreshSnippets();
   340→    } catch (err) {
   341→      const msg =
   342→        err instanceof Error ? err.message : "Gagal menyimpan snippet";
   343→      toast.error("Gagal menyimpan", { description: msg });
   344→    } finally {
   345→      setSaving(false);
   346→    }
   347→  };
   348→
   349→  // ─── Load a saved snippet into the result panel ───
   350→  // Replaces the current extracted blocks with the snippet's blocks and
   351→  // updates the filename/lang so the user can keep editing / re-saving.
   352→  const handleLoadSnippet = async (s: SnippetMeta) => {
   353→    try {
   354→      const detail = await getSnippet(s.id);
   355→      setUploadedFile(
   356→        new File([""], detail.originalFilename, { type: "text/plain" }),
   357→      );
   358→      setExtractedBlocks(detail.blocks);
   359→      setExtractMeta({ method: "saved-snippet" });
   360→      setSavedSnippetId(detail.id);
   361→      setExtractError(null);
   362→      setSelectedLang(detail.extractedLang || "auto");
   363→      toast.success("Snippet dimuat", {
   364→        description: `${detail.blocks.length} blok · ${detail.originalFilename}`,
   365→      });
   366→    } catch (err) {
   367→      const msg = err instanceof Error ? err.message : "Gagal memuat snippet";
   368→      toast.error("Gagal memuat", { description: msg });
   369→    }
   370→  };
   371→
   372→  const handleDeleteSnippet = async (s: SnippetMeta) => {
   373→    if (!confirm(`Hapus snippet "${s.originalFilename}"?`)) return;
   374→    try {
   375→      await deleteSnippet(s.id);
   376→      setSnippets((prev) => prev.filter((x) => x.id !== s.id));
   377→      toast.success("Snippet dihapus", { description: s.originalFilename });
   378→    } catch (err) {
   379→      const msg = err instanceof Error ? err.message : "Gagal menghapus";
   380→      toast.error("Gagal menghapus", { description: msg });
   381→    }
   382→  };
   383→
   384→  // ─── Copy & download ───
   385→  const handleCopy = async () => {
   386→    try {
   387→      await navigator.clipboard.writeText(displayCode);
   388→      setCopied(true);
   389→      setTimeout(() => setCopied(false), 2000);
   390→      toast.success("Kode disalin!", {
   391→        description: `${activeBlock?.lines ?? displayCode.split("\n").length} baris`,
   392→      });
   393→    } catch {
   394→      toast.error("Gagal menyalin ke clipboard");
   395→    }
   396→  };
   397→
   398→  const handleDownload = () => {
   399→    if (!activeBlock) return;
   400→    const blob = new Blob([activeBlock.code], {
   401→      type: "text/plain;charset=utf-8",
   402→    });
   403→    const url = URL.createObjectURL(blob);
   404→    const a = document.createElement("a");
   405→    a.href = url;
   406→    a.download = `codelooter_${activeBlock.lang}.${currentLang?.ext ?? "txt"}`;
   407→    a.click();
   408→    URL.revokeObjectURL(url);
   409→    toast.success("File diunduh", {
   410→      description: a.download,
   411→    });
   412→  };
   413→
   414→  return (
   415→    <div
   416→      style={{
   417→        backgroundColor: "#fef9f0",
   418→        fontFamily: "var(--font-body)",
   419→        minHeight: "100vh",
   420→        display: "flex",
   421→        flexDirection: "column",
   422→      }}
   423→    >
   424→      {/* ══ HEADER ══ */}
   425→      <header
   426→        style={{
   427→          backgroundColor: "#ffe8a3",
   428→          borderBottom: "3px solid #000",
   429→          boxShadow: "0 5px 0 #000",
   430→          position: "sticky",
   431→          top: 0,
   432→          zIndex: 50,
   433→          padding: "8px 16px",
   434→          display: "flex",
   435→          alignItems: "center",
   436→          justifyContent: "space-between",
   437→          gap: "8px",
   438→        }}
   439→      >
   440→        <div
   441→          style={{
   442→            display: "flex",
   443→            alignItems: "center",
   444→            gap: "10px",
   445→            minWidth: 0,
   446→          }}
   447→        >
   448→          <div
   449→            style={{
   450→              width: 40,
   451→              height: 40,
   452→              border: "3px solid #000",
   453→              borderRadius: "10px",
   454→              backgroundColor: "#ff6b6b",
   455→              boxShadow: "3px 3px 0 #000",
   456→              flexShrink: 0,
   457→              display: "flex",
   458→              alignItems: "center",
   459→              justifyContent: "center",
   460→            }}
   461→          >
   462→            <Code2 size={22} strokeWidth={2.5} />
   463→          </div>
   464→          <h1
   465→            style={{
   466→              fontFamily: "var(--font-display)",
   467→              fontSize: "clamp(1.5rem,5vw,2.1rem)",
   468→              letterSpacing: "0.05em",
   469→              lineHeight: 1,
   470→              textShadow: "3px 3px 0 #ff6b6b",
   471→              whiteSpace: "nowrap",
   472→              margin: 0,
   473→            }}
   474→          >
   475→            CodeLooter!
   476→          </h1>
   477→          <div
   478→            style={{
   479→              backgroundColor: "#ff6b6b",
   480→              border: "2px solid #000",
   481→              borderRadius: "6px",
   482→              padding: "1px 7px",
   483→              fontFamily: "var(--font-display)",
   484→              fontSize: "0.78rem",
   485→              color: "#fff",
   486→              boxShadow: "2px 2px 0 #000",
   487→              flexShrink: 0,
   488→            }}
   489→          >
   490→            BETA
   491→          </div>
   492→        </div>
   493→
   494→        <div
   495→          style={{
   496→            display: "flex",
   497→            alignItems: "center",
   498→            gap: "8px",
   499→            flexShrink: 0,
   500→          }}
   501→        >
   502→          {extractMeta.method && (
   503→            <Tag bg="#d4f0e4" color="#000">⚡ PATTERN</Tag>
   504→          )}
   505→          <button
   506→            onClick={() => fileInputRef.current?.click()}
   507→            style={{
   508→              backgroundColor: "#000",
   509→              color: "#ffe8a3",
   510→              border: "3px solid #000",
   511→              borderRadius: "10px",
   512→              padding: "8px 14px",
   513→              fontFamily: "var(--font-display)",
   514→              fontSize: "0.95rem",
   515→              letterSpacing: "0.05em",
   516→              cursor: "pointer",
   517→              boxShadow: "4px 4px 0 #ff6b6b",
   518→              display: "flex",
   519→              alignItems: "center",
   520→              gap: "6px",
   521→              transition: "transform 0.1s, box-shadow 0.1s",
   522→            }}
   523→            onMouseDown={(e) => {
   524→              (e.currentTarget as HTMLElement).style.transform =
   525→                "translate(2px,2px)";
   526→              (e.currentTarget as HTMLElement).style.boxShadow =
   527→                "2px 2px 0 #ff6b6b";
   528→            }}
   529→            onMouseUp={(e) => {
   530→              (e.currentTarget as HTMLElement).style.transform = "";
   531→              (e.currentTarget as HTMLElement).style.boxShadow =
   532→                "4px 4px 0 #ff6b6b";
   533→            }}
   534→          >
   535→            <Upload size={15} /> PILIH FILE
   536→          </button>
   537→        </div>
   538→      </header>
   539→
   540→      {/* ══ MAIN ══ */}
   541→      <main
   542→        style={{
   543→          padding: "16px",
   544→          maxWidth: "1400px",
   545→          margin: "0 auto",
   546→          width: "100%",
   547→          flex: 1,
   548→        }}
   549→      >
   550→        {/* STATS */}
   551→        <div className="stats-grid">
   552→          {STATS.map((s) => (
   553→            <div
   554→              key={s.label}
   555→              style={{
   556→                backgroundColor: s.color,
   557→                border: "3px solid #000",
   558→                borderRadius: "10px",
   559→                padding: "10px 14px",
   560→                boxShadow: "4px 4px 0 #000",
   561→              }}
   562→            >
   563→              <p

   565→                  fontSize: "clamp(1.2rem,4vw,1.6rem)",

   590→        {/* THREE COLUMNS */}

   616→              {/* detected langs after extraction */}
   617→              {detectedLangs.length > 0 && (
   618→                <div
   619→                  style={{
   620→                    backgroundColor: "#d4f0e4",
   621→                    border: "2px solid #000",
   622→                    borderRadius: "10px",

   625→                >

   629→                      fontWeight: 900,

   631→                      letterSpacing: "0.08em",
   632→                      marginBottom: "7px",
   633→                      marginTop: 0,
   634→                    }}
   635→                  >
   636→                    🔍 Terdeteksi
   637→                  </p>
   638→                  <div
