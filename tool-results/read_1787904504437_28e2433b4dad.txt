    19→interface ResultPanelProps {
    20→  result: ExtractResult | null;
    21→  loading: boolean;
    22→  /**
    23→   * Called after a snippet is saved as a NEW record (existing behaviour).
    24→   * Receives the freshly-created snippet id (and the tags that were sent
    25→   * with the save) so the parent can adopt them as the new
    26→   * `currentSnippetId` / `currentTags` (allowing subsequent edits to
    27→   * update the same record in-place).
    28→   */
    29→  onSaved: (snippetId?: string, tags?: string) => void;
    30→  /**
    31→   * ID of the snippet currently loaded into the panel. When set, an extra
    32→   * "Update" button is rendered next to "Simpan" so the user can push
    33→   * their edits back to the same snippet record instead of creating a
    34→   * duplicate.
    35→   */
    36→  currentSnippetId?: string;
    37→  /**
    38→   * Invoked when the user clicks "Update". The parent performs the actual
    39→   * PATCH /api/snippets/[id] call, shows a toast, and refreshes the list.
    40→   * Receives the panel's current (possibly edited) blocks so the PATCH
    41→   * payload reflects in-panel edits (merge, split, delete, duplicate,
    42→   * reorder, text edits) — not the stale `result.blocks` from the parent.
    43→   * The optional `tags` parameter forwards the panel's current tag input
    44→   * so the PATCH can also update the snippet's tags in the same round-trip.
    45→   */
    46→  onUpdateSnippet?: (id: string, blocks: CodeBlock[], tags?: string) => Promise<void> | void;
    47→  /**
    48→   * Comma-separated tags of the snippet currently loaded into the panel
    49→   * (mirrors `currentSnippetId`). When a saved snippet is selected from
    50→   * the list, the parent forwards its `tags` here so the tag input can be
    51→   * pre-populated. `undefined` for a fresh extraction (no associated
    52→   * snippet) → the input is cleared.
    53→   */
    54→  currentTags?: string;
    55→}

    57→export function ResultPanel({ result, loading, onSaved, currentSnippetId, onUpdateSnippet, currentTags }: ResultPanelProps) {
    58→  const [blocks, setBlocks] = useState<CodeBlock[] | null>(null);
    59→  const [saving, setSaving] = useState(false);
    60→  const [updating, setUpdating] = useState(false);
    61→  const [copiedAll, setCopiedAll] = useState(false);
    62→  const [copiedMd, setCopiedMd] = useState(false);
    63→  const [zipping, setZipping] = useState(false);
    64→  const [exportingHtml, setExportingHtml] = useState(false);
    65→  const [exportingJson, setExportingJson] = useState(false);
    66→  const [viewMode, setViewMode] = useState<"extracted" | "comparison">("extracted");
    67→  const [reorderMode, setReorderMode] = useState(false);
    68→  // Tag input value (comma-separated). Synced from `currentTags` whenever
    69→  // the parent loads a new snippet (or clears the panel); the user can then
    70→  // freely edit the value, which is forwarded to saveSnippet / updateSnippet.
    71→  const [tagInput, setTagInput] = useState("");
    72→
    73→  // In-block search & language filter state. The search input lives below
    74→  // the file header card and narrows the visible blocks list (the header
    75→  // stats still reflect ALL blocks). Both are reset whenever a new result
    76→  // arrives so a stale query from a previous file doesn't bleed over.
    77→  const [searchQuery, setSearchQuery] = useState("");
    78→  const [langFilter, setLangFilter] = useState("all");
    79→  // Bookmark-only filter toggle. When true the visible blocks list is
    80→  // narrowed to blocks whose `bookmarked` field is `true`. Lives next to
    81→  // the language filter dropdown so the two filters can be combined
    82→  // (e.g. "show me only my starred R blocks"). Reset on new result so a
    83→  // stale filter from a previous file doesn't bleed over.
    84→  const [bookmarkedOnly, setBookmarkedOnly] = useState(false);
    85→
    86→  // Multi-select state for bulk operations (copy / delete / ZIP). The set
    87→  // holds block `index` values — the same key used by every per-block
    88→  // handler above. We reset it whenever the underlying result changes so
    89→  // stale selections from a previous file don't bleed into the new one.
    90→  const [selectedIndices, setSelectedIndices] = useState<Set<number>>(new Set());
    91→  // Tracks the bulk-delete confirmation flow (mirrors the per-block pattern
    92→  // in CodeBlockCard: first click arms, second click within 3s commits).
    93→  const [confirmBulkDelete, setConfirmBulkDelete] = useState(false);
    94→  const [bulkZipping, setBulkZipping] = useState(false);
    95→  const [bulkCopied, setBulkCopied] = useState(false);
    96→
    97→  // File extension per language, mirroring the server-side extForLang helper.
    98→  const extForLang = (lang: string): string =>
    99→    ({
   100→      r: "R",
   101→      python: "py",
   102→      sql: "sql",
   103→      java: "java",
   104→      cpp: "cpp",
   105→      javascript: "js",
   106→      typescript: "ts",
   107→      php: "php",
   108→      kotlin: "kt",
   109→      go: "go",
   110→      rust: "rs",
   111→      bash: "sh",
   112→    }[lang] ?? "txt");
   113→
   114→  // BUGFIX: Reset local editable blocks whenever a new result arrives.
   115→  // Previously, editing a block then extracting a new file would show the
   116→  // old edited blocks instead of the fresh extraction result.
   117→  useEffect(() => {
   118→    setBlocks(null);
   119→    setViewMode("extracted");
   120→    setReorderMode(false);
   121→    // Reset the in-block search & language filter so a query from the
   122→    // previous file doesn't bleed into the newly-loaded result.
   123→    setSearchQuery("");
   124→    setLangFilter("all");
   125→    // Reset the bookmark-only filter for the same reason.
   126→    setBookmarkedOnly(false);
   127→    // Clear any active multi-select so stale selections don't bleed into
   128→    // the newly-loaded result.
   129→    setSelectedIndices(new Set());
   130→    setConfirmBulkDelete(false);
   131→    // A fresh extraction has no associated snippet → clear the tag input so
   132→    // tags from a previously-loaded snippet don't bleed into the new save.
   133→    // The dedicated `currentTags` sync effect below will overwrite this
   134→    // immediately when a saved snippet is loaded instead.
   135→    setTagInput("");
   136→  }, [result]);
   137→
   138→  // Sync the tag input from the parent's `currentTags` prop. This fires
   139→  // whenever the user selects a saved snippet (parent forwards its tags) or
   140→  // when the parent clears the panel (currentTags becomes undefined). The
   141→  // effect is intentionally keyed on `currentTags` only — not `result` — so
   142→  // an in-progress tag edit isn't wiped when a fresh extraction of a
   143→  // different file replaces `result` (the reset-on-result effect above
   144→  // already handles the clear-on-new-extraction case).
   145→  useEffect(() => {
   146→    setTagInput(currentTags ?? "");
   147→  }, [currentTags]);
   148→
   149→  // Sync local editable blocks whenever a new result arrives.
   150→  const effectiveBlocks = blocks ?? result?.blocks ?? [];
   151→
   152→  // ── In-block search & language filter ─────────────────────────────────
   153→  // Text match is a case-insensitive substring search over `b.code`. The
   154→  // language match is exact-equality against the dropdown value (or "all"
   155→  // to disable the language filter). The bookmark filter is a boolean AND
   156→  // on `b.bookmarked === true`. All three conditions are AND-combined so
   157→  // a user searching "ggplot2" with the language set to "r" and the
   158→  // bookmark filter on sees only starred R blocks whose code contains
   159→  // "ggplot2".
   160→  //
   161→  // NOTE: the file header card (total count, language distribution, line
   162→  // totals, copy/zip/save buttons) still uses `effectiveBlocks` — those
   163→  // reflect the FULL extraction, not the filtered subset. Only the visible
   164→  // blocks list renders `filteredBlocks`.
   165→  const uniqueLangs = Array.from(new Set(effectiveBlocks.map((b) => b.lang)));
   166→  const trimmedQuery = searchQuery.trim().toLowerCase();
   167→  const filteredBlocks = effectiveBlocks.filter((b) => {
   168→    const matchesText = trimmedQuery === "" || b.code.toLowerCase().includes(trimmedQuery);
   169→    const matchesLang = langFilter === "all" || b.lang === langFilter;
   170→    const matchesBookmark = !bookmarkedOnly || b.bookmarked === true;
   171→    return matchesText && matchesLang && matchesBookmark;
   172→  });
   173→  const isFiltering = trimmedQuery !== "" || langFilter !== "all" || bookmarkedOnly;
   174→
   175→  // Toggle the `bookmarked` flag on the block identified by `index`. Uses
   176→  // the same local-state mutation pattern as `handleBlockChange` — only
   177→  // the matching block is replaced (with `bookmarked` flipped), the rest
   178→  // of the array is preserved. The bookmark state persists with the
   179→  // snippet when saved because `saveSnippet` / `updateSnippet` send the
   180→  // full `effectiveBlocks` array (including `bookmarked`) as JSON, and the
   181→  // API routes round-trip the blocks JSON verbatim.
   182→  //
   183→  // The toast message reads from the block's *new* state — so we look it
   184→  // up *after* the flip is computed. `bookmarked` is treated as `false`
   185→  // when undefined, so toggling a block that has never been bookmarked
   186→  // sets it to `true`.
   187→  const handleToggleBookmark = (index: number) => {
   188→    setBlocks((prev) => {
   189→      const base = prev ?? result?.blocks ?? [];
   190→      return base.map((b) =>
   191→        b.index === index ? { ...b, bookmarked: !b.bookmarked } : b,
   192→      );
   193→    });
   194→    // Look up the current (pre-flip) state for the toast — `effectiveBlocks`
   195→    // reflects the current render and `!b.bookmarked` matches the new state.
   196→    const current = effectiveBlocks.find((b) => b.index === index);
   197→    toast.info(current && !current.bookmarked ? "Blok ditandai" : "Bookmark dihapus");
   198→  };
   199→
   200→  const handleSave = async () => {
   201→    if (!result) return;
   202→    setSaving(true);
   203→    try {
   204→      const saved = await saveSnippet(
   205→        result.filename,
   206→        effectiveBlocks,
   207→        effectiveBlocks[0]?.lang ?? "unknown",
   208→        result.size,
   209→        // Forward the tag input so the new snippet is created with the
   210→        // user's tags (comma-separated). Empty string is a no-op on the
   211→        // server (default is "").
   212→        tagInput.trim(),
   213→      );
   214→      // Record the tags into the in-memory recently-used history so the
   215→      // TagInput's autocomplete row offers them on the next save.
   216→      useTagHistory.getState().addTag(tagInput);
   217→      toast.success(`Snippet disimpan · ${saved.id.slice(0, 8)}`);
   218→      // Adopt the freshly-created snippet as the current one so subsequent
   219→      // edits can be pushed back via "Update" rather than re-saving a copy.
   220→      // Also forward the tags that were sent so the parent's `currentTags`
   221→      // state matches what was persisted (otherwise the [currentTags] sync
   222→      // effect could later overwrite an in-progress tag edit).
   223→      onSaved(saved.id, tagInput.trim());
   224→    } catch (e: any) {
   225→      toast.error(e?.message ?? "Gagal menyimpan");
   226→    } finally {
   227→      setSaving(false);
   228→    }
   229→  };
   230→
   231→  // Push the current editable blocks back to the *existing* snippet record
   232→  // (identified by `currentSnippetId`). The parent owns the actual API call
   233→  // (so it can also refresh the snippet list + show a toast) — this handler
   234→  // just toggles the loading state on the button and forwards the id.
   235→  const handleUpdate = async () => {
   236→    if (!currentSnippetId || !onUpdateSnippet) return;
   237→    if (effectiveBlocks.length === 0) return;
   238→    setUpdating(true);
   239→    try {
   240→      // Pass the panel's effective (possibly edited) blocks so the parent's
   241→      // PATCH payload reflects in-panel edits — not stale result.blocks.
   242→      // Also forward the current tagInput so the snippet's tags can be
   243→      // updated alongside the code (the parent's PATCH handler accepts an
   244→      // optional `tags` parameter).
   245→      await onUpdateSnippet(currentSnippetId, effectiveBlocks, tagInput.trim());
   246→      // Record the tags into the in-memory recently-used history so the
   247→      // TagInput's autocomplete row offers them on the next edit.
   248→      useTagHistory.getState().addTag(tagInput);
   249→    } finally {
   250→      setUpdating(false);
   251→    }
   252→  };
   253→
   254→  const handleDownloadAll = () => {
   255→    if (!result) return;
   256→    // Download all blocks as a single text blob (client-side, no snippet needed).
   257→    const dominant = effectiveBlocks[0]?.lang ?? "unknown";
   258→    const ext =
   259→      { r: "R", python: "py", sql: "sql", java: "java", cpp: "cpp", javascript: "js", typescript: "ts" }[dominant] ?? "txt";
   260→    const sep = "#".repeat(60);
   261→    const body = effectiveBlocks
   262→      .map((b, i) => `${sep}\n# Block #${b.index} | ${b.lang} | ${b.lines} lines\n${sep}\n${b.code}`)
   263→      .join("\n\n");
   264→    const blob = new Blob([body], { type: "text/plain;charset=utf-8" });
   265→    const url = URL.createObjectURL(blob);
   266→    const a = document.createElement("a");
   267→    a.href = url;
   268→    a.download = `${result.filename.replace(/\.[^.]+$/, "") || "codelooter"}.${ext}`;
   269→    a.click();
   270→    URL.revokeObjectURL(url);
   271→    toast.success("File kode diunduh");
   272→  };
   273→
   274→  const handleDownloadZip = async () => {
   275→    if (!result || effectiveBlocks.length === 0) return;
   276→    setZipping(true);
   277→    try {
   278→      // Dynamic import keeps jszip out of the initial client bundle.
   279→      const JSZip = (await import("jszip")).default;
   280→      const zip = new JSZip();
   281→      const used = new Map<string, number>();
   282→      for (const b of effectiveBlocks) {
   283→        const ext = extForLang(b.lang);
   284→        let name = `block_${b.index}.${ext}`;
   285→        if (used.has(name)) {
   286→          const n = used.get(name)! + 1;
   287→          used.set(name, n);
   288→          name = `block_${b.index}_${n}.${ext}`;
   289→        } else {
   290→          used.set(name, 0);
   291→        }
   292→        zip.file(name, b.code);
   293→      }
   294→      const blob = await zip.generateAsync({
   295→        type: "blob",
   296→        compression: "DEFLATE",
   297→        compressionOptions: { level: 6 },
   298→      });
   299→      const url = URL.createObjectURL(blob);
   300→      const a = document.createElement("a");
   301→      a.href = url;
   302→      const base = result.filename.replace(/\.[^.]+$/, "") || "codelooter";
   303→      a.download = `${base}_blocks.zip`;
   304→      a.click();
   305→      URL.revokeObjectURL(url);
   306→      toast.success(`ZIP dengan ${effectiveBlocks.length} file dibuat`);
   307→    } catch (e: any) {
   308→      toast.error(e?.message ?? "Gagal membuat ZIP");
   309→    } finally {
   310→      setZipping(false);
   311→    }
   312→  };

   314→  // Build a self-contained, syntax-highlighted HTML file with all extracted
   315→  // code blocks. Reuses the lightweight `highlight()` tokenizer from
   316→  // code-block-card so the exported colours match the in-app dark-mode view.

   319→    if (!result || effectiveBlocks.length === 0) return;
   320→    setExportingHtml(true);
   321→    try {
   322→      // HTML-escape every text node before it lands inside <code>.
   323→      const escapeHtml = (s: string): string =>
   324→        s
   325→          .replace(/&/g, "&amp;")
   326→          .replace(/</g, "&lt;")
   327→          .replace(/>/g, "&gt;")
   328→          .replace(/"/g, "&quot;")
   329→          .replace(/'/g, "&#39;");
   330→
   331→      // Inline-style palette mirroring TOKEN_CLASS dark-mode colours.
   332→      // Typed against `keyof typeof TOKEN_CLASS` so this stays in sync if a
   333→      // new token category is added upstream. Strictly emerald / teal / amber
   334→      // / rose / slate — no blue or indigo.
   335→      const HTML_COLOR: Record<keyof typeof TOKEN_CLASS, { color: string; extra?: string }> = {
   336→        comment: { color: "#94a3b8", extra: "font-style: italic;" }, // slate-400
   337→        string:  { color: "#34d399" },                              // emerald-400
   338→        number:  { color: "#fbbf24" },                              // amber-400
   339→        keyword: { color: "#fb7185", extra: "font-weight: 600;" },  // rose-400
   340→        func:    { color: "#2dd4bf" },                              // teal-400
   341→        ident:   { color: "#e2e8f0" },                              // slate-200
   342→        op:      { color: "#94a3b8" },                              // slate-400
   343→        nl:      { color: "inherit" },
   344→      };
   345→
   346→      // Render a single block's code to highlighted, escaped HTML spans.
   347→      const renderCode = (b: CodeBlock): string => {
   348→        const toks = highlight(b.code, b.lang);
   349→        return toks
   350→          .map((tk) => {
   351→            const esc = escapeHtml(tk.t);
   352→            const cfg = HTML_COLOR[tk.c] ?? HTML_COLOR.ident;
   353→            if (tk.c === "nl") return esc;
   354→            const style = cfg.extra
   355→              ? `color:${cfg.color};${cfg.extra}`
   356→              : `color:${cfg.color}`;
   357→            return `<span style="${style}">${esc}</span>`;
   358→          })
   359→          .join("");
   360→      };
   361→
   362→      const baseName = result.filename.replace(/\.[^.]+$/, "") || "codelooter";
   363→      const today = new Date().toISOString().slice(0, 10);
   364→
   365→      const sections = effectiveBlocks
   366→        .map((b) => {
   367→          const langLabel = (b.lang || "text").toUpperCase();
   368→          return `    <section class="block">
   369→      <header class="block-head">
   370→        <span class="idx">#${b.index}</span>
   371→        <span class="lang">${escapeHtml(langLabel)}</span>
   372→        <span class="meta">${b.lines} lines</span>
   373→      </header>
   374→      <pre><code>${renderCode(b)}</code></pre>
   375→    </section>`;
   376→        })
   377→        .join("\n");
   378→
   379→      const html = `<!DOCTYPE html>
   380→<html lang="en">
   381→<head>
   382→  <meta charset="UTF-8">
   383→  <meta name="viewport" content="width=device-width, initial-scale=1.0">
   384→  <title>${escapeHtml(baseName)} — CodeLooter Export</title>
   385→  <style>
   386→    :root { color-scheme: dark; }
   387→    * { box-sizing: border-box; }
   388→    body {
   389→      margin: 0;
   390→      background: #0f172a;
   391→      color: #e2e8f0;
   392→      font-family: -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, Helvetica, Arial, sans-serif;
   393→      padding: 1.5rem 1rem 3rem;
   394→      line-height: 1.5;
   395→    }
   396→    .container { max-width: 960px; margin: 0 auto; }
   397→    h1 {
   398→      font-size: 1.5rem;
   399→      font-weight: 700;
   400→      margin: 0 0 .25rem;
   401→      color: #f8fafc;
   402→      word-break: break-all;
   403→    }
   404→    .subtitle {
   405→      color: #94a3b8;
   406→      font-size: .875rem;
   407→      margin: 0 0 1.5rem;
   408→    }
   409→    .block {
   410→      background: #1e293b;
   411→      border: 1px solid #334155;
   412→      border-radius: .6rem;
   413→      margin: 0 0 1rem;
   414→      overflow: hidden;
   415→    }
   416→    .block-head {
   417→      display: flex;
   418→      align-items: center;
   419→      gap: .75rem;
   420→      padding: .6rem .9rem;
   421→      background: #0f172a;
   422→      border-bottom: 1px solid #334155;
   423→      font-size: .8rem;
   424→    }
   425→    .block-head .idx {
   426→      font-family: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;
   427→      color: #94a3b8;
   428→      font-weight: 600;
   429→    }
   430→    .block-head .lang {
   431→      background: rgba(16,185,129,.15);
   432→      color: #6ee7b7;
   433→      padding: .15rem .5rem;
   434→      border-radius: 999px;
   435→      font-size: .7rem;
   436→      font-weight: 600;
   437→      letter-spacing: .03em;
   438→    }
   439→    .block-head .meta {
   440→      color: #64748b;
   441→      font-size: .75rem;
   442→    }
   443→    pre {
   444→      margin: 0;
   445→      padding: 1rem;
   446→      overflow-x: auto;
   447→      background: #0f172a;
   448→    }
   449→    code {
   450→      font-family: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;
   451→      font-size: .82rem;
   452→      line-height: 1.6;
   453→      color: #e2e8f0;

   465→</head>
   466→<body>
   467→  <div class="container">
   468→    <h1>${escapeHtml(result.filename)}</h1>
   469→    <p class="subtitle">${effectiveBlocks.length} blok kode · ${totalLines} baris · dihasilkan oleh CodeLooter</p>
   470→${sections}
   471→    <div class="footer">Diekspor dari CodeLooter — ${today}</div>
   472→  </div>
   473→</body>
   474→</html>`;
   475→
   476→      const blob = new Blob([html], { type: "text/html;charset=utf-8" });
   477→      const url = URL.createObjectURL(blob);
   478→      const a = document.createElement("a");
   479→      a.href = url;
   480→      a.download = `${baseName}_blocks.html`;
   481→      a.click();
   482→      URL.revokeObjectURL(url);
   483→      toast.success(`HTML dengan ${effectiveBlocks.length} blok dibuat`);
   484→    } catch (e: any) {
   485→      toast.error(e?.message ?? "Gagal membuat HTML");
   486→    } finally {
   487→      setExportingHtml(false);
   488→    }
   489→  };

   491→  // Build a structured JSON export containing the filename, file size, total
   492→  // block count, an ISO timestamp of the export moment, the extraction
   493→  // stats (when available) and the full per-block payload (index, language,
   494→  // raw code, line count and the source/origin tag). The file is downloaded
   495→  // client-side as a Blob (no round-trip to the server) and named
   496→  // `${base}_export.json` so it doesn't collide with the txt/zip/html exports.
   497→  // Loading state is kept symmetrical with the other export buttons even
   498→  // though JSON serialization is effectively instant.
   499→  const handleDownloadJson = () => {
   500→    if (!result || effectiveBlocks.length === 0) return;

   502→    try {
   503→      const payload = {
   504→        filename: result.filename,
   505→        fileSize: result.size,
   506→        totalBlocks: effectiveBlocks.length,
   507→        extractedAt: new Date().toISOString(),
   508→        stats: result.stats ?? null,
   509→        blocks: effectiveBlocks.map((b) => ({
   510→          index: b.index,
   511→          lang: b.lang,
   512→          code: b.code,
   513→          lines: b.lines,
   514→          source: b.source,
   515→        })),
   516→      };

   518→      const blob = new Blob([json], { type: "application/json;charset=utf-8" });
   519→      const url = URL.createObjectURL(blob);
   520→      const a = document.createElement("a");
   521→      a.href = url;
   522→      const base = result.filename.replace(/\.[^.]+$/, "") || "codelooter";

   528→      toast.error(e?.message ?? "Gagal membuat JSON");

   534→  const handleBlockDownload = (index: number) => {

   538→      { r: "R", python: "py", sql: "sql", java: "java", cpp: "cpp", javascript: "js", typescript: "ts" }[b.lang] ?? "txt";

   575→  // Merge block at `index` with the next block. The two blocks' code is
   576→  // concatenated with a newline, and indices are renumbered.
   577→  const handleMergeWithNext = (index: number) => {
   578→    setBlocks((prev) => {
   579→      const base = prev ?? result?.blocks ?? [];
   580→      const idx = base.findIndex((b) => b.index === index);

   582→      const merged = {
   583→        ...base[idx],
   584→        code: base[idx].code + "\n" + base[idx + 1].code,
   585→        lines: base[idx].lines + base[idx + 1].lines,
   586→        source: base[idx].source === base[idx + 1].source
   587→          ? base[idx].source
   588→          : `${base[idx].source}+merge`,
   589→      };
   590→      const next = [...base.slice(0, idx), merged, ...base.slice(idx + 2)];
   591→      return next.map((b, i) => ({ ...b, index: i }));
   592→    });
   593→  };

   595→  // Split block at `index` into two blocks at line `atLine`. Lines 1..atLine-1
   596→  // stay in the first block, lines atLine..end go to a new second block.

   598→    setBlocks((prev) => {
   599→      const base = prev ?? result?.blocks ?? [];
   600→      const idx = base.findIndex((b) => b.index === index);
   601→      if (idx < 0) return base;
   602→      const lines = base[idx].code.split("\n");
   603→      if (atLine < 2 || atLine >= lines.length) return base;
   604→      const firstCode = lines.slice(0, atLine).join("\n");
   605→      const secondCode = lines.slice(atLine).join("\n");
   606→      const first = { ...base[idx], code: firstCode, lines: firstCode.split("\n").length };
   607→      const second = { ...base[idx], code: secondCode, lines: secondCode.split("\n").length, source: "split" };
   608→      const next = [...base.slice(0, idx), first, second, ...base.slice(idx + 1)];
   609→      return next.map((b, i) => ({ ...b, index: i }));
   610→    });
   611→  };
   612→

   614→  const handleDelete = (index: number) => {
   615→    setBlocks((prev) => {
   616→      const base = prev ?? result?.blocks ?? [];
   617→      const next = base.filter((b) => b.index !== index);
   618→      return next.map((b, i) => ({ ...b, index: i }));

   630→      return next.map((b, i) => ({ ...b, index: i }));
   631→    });
   632→  };
