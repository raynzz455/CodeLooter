    18→interface ResultPanelProps {
    19→  result: ExtractResult | null;
    20→  loading: boolean;
    21→  /**
    22→   * Called after a snippet is saved as a NEW record (existing behaviour).
    23→   * Receives the freshly-created snippet id (and the tags that were sent
    24→   * with the save) so the parent can adopt them as the new
    25→   * `currentSnippetId` / `currentTags` (allowing subsequent edits to
    26→   * update the same record in-place).
    27→   */
    28→  onSaved: (snippetId?: string, tags?: string) => void;
    29→  /**
    30→   * ID of the snippet currently loaded into the panel. When set, an extra
    31→   * "Update" button is rendered next to "Simpan" so the user can push
    32→   * their edits back to the same snippet record instead of creating a
    33→   * duplicate.
    34→   */
    35→  currentSnippetId?: string;
    36→  /**
    37→   * Invoked when the user clicks "Update". The parent performs the actual
    38→   * PATCH /api/snippets/[id] call, shows a toast, and refreshes the list.
    39→   * Receives the panel's current (possibly edited) blocks so the PATCH
    40→   * payload reflects in-panel edits (merge, split, delete, duplicate,
    41→   * reorder, text edits) — not the stale `result.blocks` from the parent.
    42→   * The optional `tags` parameter forwards the panel's current tag input
    43→   * so the PATCH can also update the snippet's tags in the same round-trip.
    44→   */
    45→  onUpdateSnippet?: (id: string, blocks: CodeBlock[], tags?: string) => Promise<void> | void;
    46→  /**
    47→   * Comma-separated tags of the snippet currently loaded into the panel
    48→   * (mirrors `currentSnippetId`). When a saved snippet is selected from
    49→   * the list, the parent forwards its `tags` here so the tag input can be
    50→   * pre-populated. `undefined` for a fresh extraction (no associated
    51→   * snippet) → the input is cleared.
    52→   */
    53→  currentTags?: string;
    54→}

    56→export function ResultPanel({ result, loading, onSaved, currentSnippetId, onUpdateSnippet, currentTags }: ResultPanelProps) {
    57→  const [blocks, setBlocks] = useState<CodeBlock[] | null>(null);
    58→  const [saving, setSaving] = useState(false);
    59→  const [updating, setUpdating] = useState(false);
    60→  const [copiedAll, setCopiedAll] = useState(false);
    61→  const [copiedMd, setCopiedMd] = useState(false);
    62→  const [zipping, setZipping] = useState(false);
    63→  const [exportingHtml, setExportingHtml] = useState(false);
    64→  const [exportingJson, setExportingJson] = useState(false);
    65→  const [viewMode, setViewMode] = useState<"extracted" | "comparison">("extracted");
    66→  const [reorderMode, setReorderMode] = useState(false);
    67→  // Tag input value (comma-separated). Synced from `currentTags` whenever
    68→  // the parent loads a new snippet (or clears the panel); the user can then
    69→  // freely edit the value, which is forwarded to saveSnippet / updateSnippet.
    70→  const [tagInput, setTagInput] = useState("");
    71→
    72→  // In-block search & language filter state. The search input lives below

    74→  // stats still reflect ALL blocks). Both are reset whenever a new result
    75→  // arrives so a stale query from a previous file doesn't bleed over.

    77→  const [langFilter, setLangFilter] = useState("all");

    79→  // Multi-select state for bulk operations (copy / delete / ZIP). The set

    82→  // stale selections from a previous file don't bleed into the new one.
    83→  const [selectedIndices, setSelectedIndices] = useState<Set<number>>(new Set());

    87→  const [bulkZipping, setBulkZipping] = useState(false);
    88→  const [bulkCopied, setBulkCopied] = useState(false);

    90→  // File extension per language, mirroring the server-side extForLang helper.
    91→  const extForLang = (lang: string): string =>
    92→    ({
    93→      r: "R",
    94→      python: "py",
    95→      sql: "sql",
    96→      java: "java",
    97→      cpp: "cpp",
    98→      javascript: "js",
    99→      typescript: "ts",
   100→      php: "php",

   117→    setLangFilter("all");
   118→    // Clear any active multi-select so stale selections don't bleed into

   122→    // A fresh extraction has no associated snippet → clear the tag input so
   123→    // tags from a previously-loaded snippet don't bleed into the new save.
   124→    // The dedicated `currentTags` sync effect below will overwrite this
   125→    // immediately when a saved snippet is loaded instead.
   126→    setTagInput("");

   130→  // whenever the user selects a saved snippet (parent forwards its tags) or
   131→  // when the parent clears the panel (currentTags becomes undefined). The
   132→  // effect is intentionally keyed on `currentTags` only — not `result` — so
   133→  // an in-progress tag edit isn't wiped when a fresh extraction of a

   136→  useEffect(() => {
   137→    setTagInput(currentTags ?? "");
   138→  }, [currentTags]);
   139→
   140→  // Sync local editable blocks whenever a new result arrives.
   141→  const effectiveBlocks = blocks ?? result?.blocks ?? [];

   143→  // ── In-block search & language filter ─────────────────────────────────
   144→  // Text match is a case-insensitive substring search over `b.code`. The
   145→  // language match is exact-equality against the dropdown value (or "all"

   150→  // NOTE: the file header card (total count, language distribution, line

   154→  const uniqueLangs = Array.from(new Set(effectiveBlocks.map((b) => b.lang)));

   157→    const matchesText = trimmedQuery === "" || b.code.toLowerCase().includes(trimmedQuery);
   158→    const matchesLang = langFilter === "all" || b.lang === langFilter;
   159→    return matchesText && matchesLang;
   160→  });
   161→  const isFiltering = trimmedQuery !== "" || langFilter !== "all";
   162→
   163→  const handleSave = async () => {
   164→    if (!result) return;
   165→    setSaving(true);
   166→    try {
   167→      const saved = await saveSnippet(
   168→        result.filename,
   169→        effectiveBlocks,
   170→        effectiveBlocks[0]?.lang ?? "unknown",
   171→        result.size,
   172→        // Forward the tag input so the new snippet is created with the
   173→        // user's tags (comma-separated). Empty string is a no-op on the

   179→      // edits can be pushed back via "Update" rather than re-saving a copy.
   180→      // Also forward the tags that were sent so the parent's `currentTags`
   181→      // state matches what was persisted (otherwise the [currentTags] sync
   182→      // effect could later overwrite an in-progress tag edit).
   183→      onSaved(saved.id, tagInput.trim());
   184→    } catch (e: any) {
   185→      toast.error(e?.message ?? "Gagal menyimpan");
   186→    } finally {
   187→      setSaving(false);
   188→    }
   189→  };
   190→
   191→  // Push the current editable blocks back to the *existing* snippet record
   192→  // (identified by `currentSnippetId`). The parent owns the actual API call

   203→      // updated alongside the code (the parent's PATCH handler accepts an

   205→      await onUpdateSnippet(currentSnippetId, effectiveBlocks, tagInput.trim());

   207→      setUpdating(false);
   208→    }

   211→  const handleDownloadAll = () => {

   223→    const a = document.createElement("a");

   231→  const handleDownloadZip = async () => {
   232→    if (!result || effectiveBlocks.length === 0) return;
   233→    setZipping(true);
   234→    try {
   235→      // Dynamic import keeps jszip out of the initial client bundle.
   236→      const JSZip = (await import("jszip")).default;
   237→      const zip = new JSZip();
   238→      const used = new Map<string, number>();

   271→  // Build a self-contained, syntax-highlighted HTML file with all extracted

   283→          .replace(/</g, "&lt;")
   284→          .replace(/>/g, "&gt;")
   285→          .replace(/"/g, "&quot;")
   286→          .replace(/'/g, "&#39;");
   287→
   288→      // Inline-style palette mirroring TOKEN_CLASS dark-mode colours.
   289→      // Typed against `keyof typeof TOKEN_CLASS` so this stays in sync if a
   290→      // new token category is added upstream. Strictly emerald / teal / amber
   291→      // / rose / slate — no blue or indigo.
   292→      const HTML_COLOR: Record<keyof typeof TOKEN_CLASS, { color: string; extra?: string }> = {

   296→        keyword: { color: "#fb7185", extra: "font-weight: 600;" },  // rose-400
   297→        func:    { color: "#2dd4bf" },                              // teal-400
   298→        ident:   { color: "#e2e8f0" },                              // slate-200
   299→        op:      { color: "#94a3b8" },                              // slate-400
   300→        nl:      { color: "inherit" },
   301→      };
   302→
   303→      // Render a single block's code to highlighted, escaped HTML spans.
   304→      const renderCode = (b: CodeBlock): string => {
   305→        const toks = highlight(b.code, b.lang);
   306→        return toks
   307→          .map((tk) => {
   308→            const esc = escapeHtml(tk.t);
   309→            const cfg = HTML_COLOR[tk.c] ?? HTML_COLOR.ident;
   310→            if (tk.c === "nl") return esc;
   311→            const style = cfg.extra
   312→              ? `color:${cfg.color};${cfg.extra}`
   313→              : `color:${cfg.color}`;
   314→            return `<span style="${style}">${esc}</span>`;
   315→          })
   316→          .join("");
   317→      };
   318→
   319→      const baseName = result.filename.replace(/\.[^.]+$/, "") || "codelooter";

   321→
   322→      const sections = effectiveBlocks
   323→        .map((b) => {

   326→      <header class="block-head">
   327→        <span class="idx">#${b.index}</span>
   328→        <span class="lang">${escapeHtml(langLabel)}</span>
   329→        <span class="meta">${b.lines} lines</span>
   330→      </header>
   331→      <pre><code>${renderCode(b)}</code></pre>
   332→    </section>`;
   333→        })
   334→        .join("\n");

   336→      const html = `<!DOCTYPE html>
   337→<html lang="en">

   360→    }

   365→    }
   366→    .block {

   368→      border: 1px solid #334155;
   369→      border-radius: .6rem;
   370→      margin: 0 0 1rem;
   371→      overflow: hidden;
   372→    }
   373→    .block-head {
   374→      display: flex;
   375→      align-items: center;

   385→      font-weight: 600;

   387→    .block-head .lang {

   429→  </div>
   430→</body>
   431→</html>`;
   432→
   433→      const blob = new Blob([html], { type: "text/html;charset=utf-8" });
   434→      const url = URL.createObjectURL(blob);
   435→      const a = document.createElement("a");
   436→      a.href = url;

   448→  // Build a structured JSON export containing the filename, file size, total
   449→  // block count, an ISO timestamp of the export moment, the extraction
   450→  // stats (when available) and the full per-block payload (index, language,
   451→  // raw code, line count and the source/origin tag). The file is downloaded
   452→  // client-side as a Blob (no round-trip to the server) and named
   453→  // `${base}_export.json` so it doesn't collide with the txt/zip/html exports.
   454→  // Loading state is kept symmetrical with the other export buttons even

   456→  const handleDownloadJson = () => {

   481→      a.click();

   491→  const handleBlockDownload = (index: number) => {

   495→      { r: "R", python: "py", sql: "sql", java: "java", cpp: "cpp", javascript: "js", typescript: "ts" }[b.lang] ?? "txt";

   503→  };

   533→  // concatenated with a newline, and indices are renumbered.
   534→  const handleMergeWithNext = (index: number) => {

   538→      if (idx < 0 || idx >= base.length - 1) return base;

   540→        ...base[idx],
   541→        code: base[idx].code + "\n" + base[idx + 1].code,
   542→        lines: base[idx].lines + base[idx + 1].lines,
   543→        source: base[idx].source === base[idx + 1].source
   544→          ? base[idx].source
   545→          : `${base[idx].source}+merge`,
   546→      };
   547→      const next = [...base.slice(0, idx), merged, ...base.slice(idx + 2)];
   548→      return next.map((b, i) => ({ ...b, index: i }));
   549→    });
   550→  };

   552→  // Split block at `index` into two blocks at line `atLine`. Lines 1..atLine-1
   553→  // stay in the first block, lines atLine..end go to a new second block.

   555→    setBlocks((prev) => {
   556→      const base = prev ?? result?.blocks ?? [];
   557→      const idx = base.findIndex((b) => b.index === index);
   558→      if (idx < 0) return base;
   559→      const lines = base[idx].code.split("\n");
   560→      if (atLine < 2 || atLine >= lines.length) return base;
   561→      const firstCode = lines.slice(0, atLine).join("\n");
   562→      const secondCode = lines.slice(atLine).join("\n");
   563→      const first = { ...base[idx], code: firstCode, lines: firstCode.split("\n").length };
   564→      const second = { ...base[idx], code: secondCode, lines: secondCode.split("\n").length, source: "split" };
   565→      const next = [...base.slice(0, idx), first, second, ...base.slice(idx + 1)];
   566→      return next.map((b, i) => ({ ...b, index: i }));
   567→    });
   568→  };
   569→

   571→  const handleDelete = (index: number) => {
   572→    setBlocks((prev) => {
   573→      const base = prev ?? result?.blocks ?? [];
   574→      const next = base.filter((b) => b.index !== index);
   575→      return next.map((b, i) => ({ ...b, index: i }));
   576→    });
   577→  };

  1059→      {/* Blocks or Comparison view */}
